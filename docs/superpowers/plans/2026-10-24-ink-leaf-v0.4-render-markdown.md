# 墨页 v0.4 Code Highlighting and Markdown Completion Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有三端共享编辑器中加入离线代码高亮，并补齐 Setext 标题、缩进代码块、裸 URL 自动链接和 ATX 闭合井号，同时保持 Markdown 原文 round-trip 与现有安全边界。

**Architecture:** 代码块复用现有 `.rich-block` 双态：非激活显示高亮预览，激活显示原始源码；高亮片段通过 markdown-it 占位符在主消毒后回填。Markdown 补齐优先扩展 `web/model.js` 的块模型和 `renderInline()`，不引入文档级渲染或原生端改动。

**Tech Stack:** 原生 ES modules、markdown-it、DOMPurify、KaTeX、highlight.js、Node test、Playwright、Electron、Android Gradle。

**Spec:** `docs/superpowers/specs/2026-10-24-ink-leaf-v0.4-render-markdown-design.md`

## Global Constraints

- 运行时不访问网络；highlight.js 必须随 `web/vendor/renderer.js` 离线打包，不引入 CDN。
- 用户 HTML 的 DOMPurify 配置不得放宽：继续禁止 `img`、`src`、`style`、`class`、`data-*` 和未知 URI。
- 所有 Markdown 原文必须 round-trip；保存后的文本与输入一致，既有的 BOM/CRLF 处理不变。
- 单块代码高亮上限为 `100_000` 字符，超限回退纯文本。
- 不修改 `web/app.js` 的 localStorage key：`inkstone.desktop.draft.v03` / `inkstone.draft.v03` 保持不变，避免旧草稿丢失。
- 不新增原生 IPC、Android 权限、文件桥或平台构建；Android 只重新同步 `web/` 资源。
- 不实现图片、脚注、链接引用定义、定义列表、表格 WYSIWYG、自动保存、PDF、多标签。
- 版本提升到 `0.4.0`；不宣称完整 CommonMark，不宣称真机已验证。
- 每个任务结束前必须运行该任务列出的测试；最终 PR 记录全部命令与结果。

```js
const MAX_HIGHLIGHT_CHARS = 100_000;

function plainFence(content) {
  return `<pre><code>${md.utils.escapeHtml(content)}</code></pre>\n`;
}

function sanitizeHighlight(value) {
  const clean = DOMPurify.sanitize(value, { ALLOWED_TAGS: ['span'], ALLOWED_ATTR: ['class'] });
  return clean.replace(/class="([^"]*)"/g, (_, classes) => {
    const kept = classes.split(/\s+/).filter((name) => name === 'hljs' || name.startsWith('hljs-'));
    return kept.length ? ` class="${kept.join(' ')}"` : '';
  });
}

md.renderer.rules.fence = (tokens, index, options, env) => {
  const token = tokens[index];
  const info = token.info.trim().split(/\s+/)[0] || '';
  if (!env?.inkHighlight || !info || token.content.length > MAX_HIGHLIGHT_CHARS || !hljs.getLanguage(info)) {
    return plainFence(token.content);
  }
  let fragment;
  try {
    fragment = sanitizeHighlight(hljs.highlight(token.content, { language: info, ignoreIllegals: true }).value);
  } catch {
    return plainFence(token.content);
  }
  const id = env.inkCodeFragments.length;
  env.inkCodeFragments.push(fragment);
  return `<pre><code>${env.inkCodeSeed}${id}END</code></pre>\n`;
};
```

```js
export function previewMarkdown(raw, type = 'paragraph', options = {}) {
  const formulas = [];
  const codeFragments = [];
  const seed = `INKMATH${crypto.randomUUID().replaceAll('-', '')}TOKEN`;
  const codeSeed = `INKCODE${crypto.randomUUID().replaceAll('-', '')}TOKEN`;
  const env = {
    inkHighlight: options.highlight !== false,
    inkCodeFragments: codeFragments,
    inkCodeSeed: codeSeed,
  };
  // ... keep the existing formula function unchanged ...
  let html;
  if (type === 'math') {
    // ... keep existing math branch unchanged ...
  } else if (type === 'hr') html = '<hr>';
  else html = md.render(raw, env);
  let clean = DOMPurify.sanitize(html, {
    USE_PROFILES: { html: true },
    FORBID_TAGS: ['style','script','iframe','object','embed','form','input','button','textarea','select','option','img','video','audio','source','link','meta','base'],
    FORBID_ATTR: ['style','class','id','name','src','srcset','contenteditable','tabindex','autofocus'],
    ALLOW_DATA_ATTR: false,
    ADD_URI_SAFE_ATTR: ['align', 'colspan', 'rowspan', 'title'],
    ALLOWED_URI_REGEXP: /^(?:https?:|mailto:|#[^\s]*$)/i,
  });
  formulas.forEach((value, index) => { clean = clean.replaceAll(`${seed}${index}END`, value); });
  codeFragments.forEach((value, index) => { clean = clean.replaceAll(`${codeSeed}${index}END`, value); });
  return clean || '<span class="safe-empty">此 HTML 的不安全内容已隐藏，点击可编辑原文。</span>';
}
```

```js
function decorate(block) {
  const raw = block.raw;
  if (needsPreview(block)) {
    let previewRaw = raw, checkbox = '';
    if (block.type === 'list') {
      const task = raw.match(/^(\s*(?:[-+*]|\d+[.)])\s+)\[([ xX])\]\s/);
      previewRaw = raw.replace(/^\s*(?:[-+*]|\d+[.)])[ \t]+(?:\[[ xX]\][ \t]+)?/, '');
      if (task) checkbox = `<button class="task-box" contenteditable="false" tabindex="0" role="checkbox" aria-checked="${task[2].toLowerCase() === 'x'}" aria-label="切换任务完成状态" data-task="${block.id}"></button>`;
    }
    if (block.type === 'heading') previewRaw = raw.replace(/^ {0,3}#{1,6}[ \t]+/, '');
    if (block.type === 'quote') previewRaw = raw.replace(/^ {0,3}>[ \t]?/gm, '');
    const highlight = !(block.type === 'code' && block.id === activeId);
    return `${checkbox}<span class="block-source">${esc(raw)}</span><div class="block-preview task-body" contenteditable="false">${previewMarkdown(previewRaw, block.type, { highlight })}</div>`;
  }
  let prefix = '';
  if (block.type === 'heading') prefix = raw.match(/^ {0,3}#{1,6}(?:[ \t]+|$)/)?.[0] || '';
  if (block.type === 'list') prefix = raw.match(/^\s*(?:[-+*]|\d+[.)])[ \t]+/)?.[0] || '';
  if (block.type === 'quote') prefix = raw.match(/^ {0,3}> ?/)?.[0] || '';
  const task = block.type === 'list' && raw.match(/^(\s*(?:[-+*]|\d+[.)])\s+)\[([ xX])\]\s/);
  if (task) {
    prefix = task[0];
    const checked = task[2].toLowerCase() === 'x';
    return `<button class="task-box" contenteditable="false" tabindex="0" role="checkbox" aria-checked="${checked}" aria-label="切换任务完成状态" data-task="${block.id}"></button><span class="syntax">${esc(prefix)}</span><span class="task-body">${renderInline(raw.slice(prefix.length))}</span>`;
  }
  return (prefix ? `<span class="syntax">${esc(prefix)}</span>` : '') + renderInline(raw.slice(prefix.length));
}
```

```js
function trimUrlTail(url) {
  let value = url;
  while (/[.,;:!?。，；：！？]$/u.test(value)) value = value.slice(0, -1);
  while (value.endsWith(')')) {
    const opens = (value.match(/\(/g) || []).length;
```
